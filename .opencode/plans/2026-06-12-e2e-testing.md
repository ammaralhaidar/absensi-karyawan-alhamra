**Approach:** Playwright with 3 workers, headless mode. Each test uses real Supabase accounts. Camera scanning is mocked via token injection. Test data is cleaned up after each test.

**Tech Stack:** Playwright, @playwright/test, TypeScript

### Test 3: Kiosk Scanner (Check-in & Check-out)

Create: `tests/e2e/kiosk-scan.spec.ts`

```typescript
import { test, expect } from "@playwright/test";
import { loginAs, accounts } from "./utils/auth";
import { cleanupTestData, getEmployeeIdByEmail, generateQRToken, supabase } from "./utils/supabase";

test.describe("Kiosk Scanner Flow", () => {
  test.beforeEach(async () => {
    await cleanupTestData();
  });

  test("Check-in via API (camera mocked)", async () => {
    const empId = await getEmployeeIdByEmail(accounts.employee.email);
    const token = await generateQRToken(empId, "check_in");
    
    // Validate via API
    const resp = await fetch("http://localhost:3000/api/qr-validate", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ token, kiosk_id: "kiosk-test" }),
    });
    const data = await resp.json();
    expect(data.success).toBe(true);
    expect(data.status).toBe("tepat_waktu");
    
    // Verify attendance_log created
    const { data: logs } = await supabase
      .from("attendance_logs")
      .select("*")
      .eq("employee_id", empId)
      .eq("date", new Date().toISOString().split("T")[0])
      .single();
    expect(logs).toBeTruthy();
    expect(logs.check_in).toBeTruthy();
    expect(logs.check_out).toBeNull();
  });

  test("Check-out after check-in", async () => {
    const empId = await getEmployeeIdByEmail(accounts.employee.email);
    // First check-in
    const token1 = await generateQRToken(empId, "check_in");
    await fetch("http://localhost:3000/api/qr-validate", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ token: token1, kiosk_id: "kiosk-test" }),
    });
    
    // Then check-out
    const token2 = await generateQRToken(empId, "check_out");
    const resp = await fetch("http://localhost:3000/api/qr-validate", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ token: token2, kiosk_id: "kiosk-test" }),
    });
    const data = await resp.json();
    expect(data.success).toBe(true);
    expect(data.status).toBe("check_out");
    
    // Verify check_out is set
    const { data: log } = await supabase
      .from("attendance_logs")
      .select("check_out")
      .eq("employee_id", empId)
      .eq("date", new Date().toISOString().split("T")[0])
      .single();
    expect(log.check_out).toBeTruthy();
  });

  test("Duplicate check-out is rejected", async () => {
    const empId = await getEmployeeIdByEmail(accounts.employee.email);
    // Check-in and check-out
    const token1 = await generateQRToken(empId, "check_in");
    await fetch("http://localhost:3000/api/qr-validate", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ token: token1, kiosk_id: "kiosk-test" }),
    });
    const token2 = await generateQRToken(empId, "check_out");
    await fetch("http://localhost:3000/api/qr-validate", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ token: token2, kiosk_id: "kiosk-test" }),
    });
    
    // Try again
    const token3 = await generateQRToken(empId, "check_out");
    const resp = await fetch("http://localhost:3000/api/qr-validate", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ token: token3, kiosk_id: "kiosk-test" }),
    });
    const data = await resp.json();
    expect(data.error).toBe("Already checked in and out today");
  });
});
```

## Step 6: Troubleshooting Playwright Stuck

If Playwright gets stuck:
1. **Hard refresh**: Use `page.reload()` with `force: true`
2. **Increase timeout**: Set `timeout: 30000` in test config
3. **Check for hydration**: Add `waitForLoadState('networkidle')` after navigation
4. **Use `waitFor`**: Instead of `expect` for dynamic content
5. **Debug mode**: Use `--debug` flag or `--ui` mode

Key fixes for Next.js 16:
- Add `await page.waitForURL()` after navigation
- Use `data-testid` attributes for stable selectors
- Wait for network idle before assertions
